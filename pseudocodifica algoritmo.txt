function lookaround()
{
    dir = 0
    leggi da sensore profondità dritto
    se il sensore di profondità rileva distanza <= 7 
    {
        assegna a dritto il valore none
        attiva la telecamera
        se rileva lettera
        {
            accendi motore per rilascio medikit
        }
        se rileva colore
        {
            aziona display
            stampa su display il colore
        }
        spegni telecamera
    }
    se il sensore di profondità rileva distanza > 7
    {
        SE dir == 0:
            Cella dritto (x, y+1)
        ALTRIMENTI SE dir = 1:
            Cella dritto (x+1, y)
        ALTRIMENTI SE dir = 2:
            Cella dritto (x-1, y)  
    }
    spegni sensore di profondità

    aziona servo
    gira di 90° a destra
    spegni servo

    leggi da sensore profondità dritto

        ALTRIMENTI SE dir = 1:
            Cella dritto (x+1, y)
        ALTRIMENTI SE dir = 2:
            Cella dritto (x-1, y)  
    }
    spegni sensore di profondità

    aziona servo
    gira di 90° a destra
    spegni servo

    attiva sensore di profondità
    se il sensore di profondità rileva un muro 
    {
        assegna al muro valore 1
        attiva la telecamera
        aziona sensore di colore
        se rileva lettera
        {
            accendi motore per rilascio medikit
            break
        }
        se rileva colore
        {
            aziona display
            stampa su display colore
            break
        }
        spegni telecamera
        spegni sensore di colore
    }
    se il sensore di profondità non rileva un muro 
    {
        assegna al muro valore 0 
    }
    spegni sensore di profondità
    comunica direzioni possibili
    assegna a cellaVisitata[][] valore i++
    scegliDirezione();
