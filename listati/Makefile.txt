# Makefile per setup e test PiCamera Raspberry Pi 4

update-system:
	sudo apt update
	sudo apt upgrade -y
	sudo apt autoremove -y

install-base-packages:
	sudo apt install git wget curl build-essential -y

install-imaging-libs:
	sudo apt install libatlas-base-dev libjasper-dev libtiff5 libjasper1 -y
	sudo apt install libharfbuzz0b libwebp6 libtiff5 libjasper1 -y

install-python-env:
	sudo apt install python3 python3-pip python3-dev python3-venv -y

setup-venv:
	mkdir -p yoloproject
	cd yoloproject && python3 -m venv venv
	@echo "Ambiente virtuale creato in yoloproject/venv"

activate-venv:
	@echo "Per attivare l'ambiente virtuale: source yoloproject/venv/bin/activate"

install-python-deps:
	source yoloproject/venv/bin/activate && pip install --upgrade pip setuptools wheel
	source yoloproject/venv/bin/activate && pip install ultralytics opencv-python pillow numpy matplotlib

install-picamera-libs:
	sudo apt install -y python3-picamera2 python3-libcamera python3-kms libcamera-tools libcamera-dev

test-camera-preview:
	libcamera-hello --timeout 5000

test-camera-photo:
	libcamera-still -o test_photo.jpg

.PHONY: update-system install-base-packages install-imaging-libs install-python-env setup-venv activate-venv install-python-deps install-picamera-libs test-camera-preview test-camera-photo
